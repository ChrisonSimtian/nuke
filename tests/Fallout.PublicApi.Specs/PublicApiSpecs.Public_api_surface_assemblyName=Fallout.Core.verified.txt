    public enum ExecutionStatus
    {
        None = 0,
        Scheduled = 1,
        NotRun = 2,
        Skipped = 3,
        Succeeded = 4,
        Failed = 5,
        Running = 6,
        Aborted = 7,
        Collective = 8,
    }

    {
        string Description { get; }

        string Name { get; }
        System.Collections.Generic.IReadOnlyCollection<string> OrderDependencyNames { get; }
        Fallout.Common.Execution.ExecutionStatus Status { get; }
        System.Collections.Generic.IReadOnlyCollection<string> TriggerNames { get; }
    }
